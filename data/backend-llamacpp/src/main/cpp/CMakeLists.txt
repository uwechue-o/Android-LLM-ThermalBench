# Native half of the llama.cpp adapter.
#
# Built only when the project is configured with -Pthermalbench.nativeLlama=true
# and -Pthermalbench.llamaCppDir=<path to a llama.cpp checkout>.
#
#   git clone --depth 1 --branch b11879 https://github.com/ggml-org/llama.cpp ~/src/llama.cpp
#   ./gradlew :data:backend-llamacpp:assembleDebug \
#       -Pthermalbench.nativeLlama=true -Pthermalbench.llamaCppDir=$HOME/src/llama.cpp
cmake_minimum_required(VERSION 3.22.1)
project(thermalbench_llama CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT DEFINED LLAMA_CPP_DIR OR LLAMA_CPP_DIR STREQUAL "")
    message(FATAL_ERROR
        "LLAMA_CPP_DIR is not set. Pass -Pthermalbench.llamaCppDir=<path to llama.cpp checkout>.")
endif()
if(NOT EXISTS "${LLAMA_CPP_DIR}/include/llama.h")
    message(FATAL_ERROR
        "LLAMA_CPP_DIR (${LLAMA_CPP_DIR}) is not a llama.cpp checkout: include/llama.h is missing.")
endif()

# Release-grade flags: a benchmark built with debug codegen measures the build,
# not the runtime.
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG")
set(CMAKE_C_FLAGS_RELEASE "-O3 -DNDEBUG")

# Server, tools and tests are not needed and cost build time.
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)

# Optional Vulkan GPU backend (-Pthermalbench.llamaVulkan=true). The NDK ships
# the C Vulkan headers, the loader and glslc, but not the C++ (Vulkan-Hpp) or
# SPIR-V headers ggml-vulkan needs, so both come from pinned Khronos checkouts.
# ggml loads Vulkan entry points dynamically, so headers newer than the NDK's
# are safe against the device loader.
option(THERMALBENCH_VULKAN "Build llama.cpp's Vulkan backend" OFF)
if(THERMALBENCH_VULKAN)
    foreach(_dir VULKAN_HEADERS_DIR SPIRV_HEADERS_DIR)
        if(NOT DEFINED ${_dir} OR ${_dir} STREQUAL "")
            message(FATAL_ERROR "THERMALBENCH_VULKAN needs ${_dir}.")
        endif()
    endforeach()
    if(NOT EXISTS "${VULKAN_HEADERS_DIR}/include/vulkan/vulkan.hpp")
        message(FATAL_ERROR "VULKAN_HEADERS_DIR (${VULKAN_HEADERS_DIR}) is not a Vulkan-Headers checkout.")
    endif()
    if(NOT EXISTS "${SPIRV_HEADERS_DIR}/include/spirv/unified1/spirv.hpp")
        message(FATAL_ERROR "SPIRV_HEADERS_DIR (${SPIRV_HEADERS_DIR}) is not a SPIRV-Headers checkout.")
    endif()

    # Ahead of the NDK sysroot, so vulkan.h and vulkan.hpp agree on a version.
    include_directories(BEFORE
        "${VULKAN_HEADERS_DIR}/include"
        "${SPIRV_HEADERS_DIR}/include")
    set(Vulkan_INCLUDE_DIR "${VULKAN_HEADERS_DIR}/include" CACHE PATH "" FORCE)
    file(GLOB _glslc "${ANDROID_NDK}/shader-tools/*/glslc*")
    set(Vulkan_GLSLC_EXECUTABLE "${_glslc}" CACHE FILEPATH "" FORCE)

    # ggml-vulkan only asks for the package to exist; the include path above is
    # what it actually uses. A one-line config stands in for an install.
    set(_spirv_config "${CMAKE_CURRENT_BINARY_DIR}/spirv-headers-config")
    file(WRITE "${_spirv_config}/SPIRV-HeadersConfig.cmake"
        "# Generated by ThermalBench: headers come from ${SPIRV_HEADERS_DIR}\n")
    set(SPIRV-Headers_DIR "${_spirv_config}" CACHE PATH "" FORCE)

    # A machine with its own host toolchain passes THERMALBENCH_HOST_TOOLCHAIN.
    if(DEFINED THERMALBENCH_HOST_TOOLCHAIN AND NOT THERMALBENCH_HOST_TOOLCHAIN STREQUAL "")
        set(GGML_VULKAN_SHADERS_GEN_TOOLCHAIN "${THERMALBENCH_HOST_TOOLCHAIN}" CACHE FILEPATH "" FORCE)
    elseif(CMAKE_HOST_WIN32)
        # The generator's own CMake run inherits the Ninja generator but not
        # its path, which the SDK's CMake knows and a bare PATH does not.
        set(_host_toolchain "${CMAKE_CURRENT_BINARY_DIR}/msvc-host-toolchain.cmake")
        file(TO_CMAKE_PATH "${CMAKE_MAKE_PROGRAM}" _ninja)
        file(WRITE "${_host_toolchain}"
            "set(CMAKE_MAKE_PROGRAM \"${_ninja}\" CACHE FILEPATH \"\" FORCE)\n"
            "include(\"${CMAKE_CURRENT_SOURCE_DIR}/cmake/msvc-host-toolchain.cmake\")\n")
        set(GGML_VULKAN_SHADERS_GEN_TOOLCHAIN "${_host_toolchain}" CACHE FILEPATH "" FORCE)
    endif()
    set(GGML_VULKAN ON CACHE BOOL "" FORCE)
    message(STATUS "Building llama.cpp with Vulkan (glslc: ${Vulkan_GLSLC_EXECUTABLE})")
else()
    set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
endif()

# Optional OpenCL GPU backend (-Pthermalbench.llamaOpenCl=true), llama.cpp's
# path tuned for Qualcomm Adreno. Only the Khronos headers are needed here: the
# OpenCL library itself is the GPU vendor's, on the device. Linking needs a
# libOpenCL.so to resolve against, so one is generated with an entry point for
# every function the pinned cl.h declares. That stub is excluded from the APK
# (see the app's packaging rules); on a device the vendor's library is loaded.
option(THERMALBENCH_OPENCL "Build llama.cpp's OpenCL backend" OFF)
if(THERMALBENCH_OPENCL)
    if(NOT DEFINED OPENCL_HEADERS_DIR OR NOT EXISTS "${OPENCL_HEADERS_DIR}/CL/cl.h")
        message(FATAL_ERROR "THERMALBENCH_OPENCL needs OPENCL_HEADERS_DIR, an OpenCL-Headers checkout.")
    endif()
    file(READ "${OPENCL_HEADERS_DIR}/CL/cl.h" _cl_header)
    string(REGEX MATCHALL "CL_API_CALL[ \t\r\n]+cl[A-Za-z0-9_]+[ \t]*\\(" _cl_decls "${_cl_header}")
    string(CONCAT _cl_stub
        "// Generated by ThermalBench from ${OPENCL_HEADERS_DIR}/CL/cl.h.\n"
        "// Link-time only; never packaged. Traps if it is ever loaded.\n")
    foreach(_decl IN LISTS _cl_decls)
        string(REGEX REPLACE "CL_API_CALL[ \t\r\n]+(cl[A-Za-z0-9_]+).*" "\\1" _fn "${_decl}")
        string(APPEND _cl_stub "extern \"C\" void ${_fn}() { __builtin_trap(); }\n")
    endforeach()
    # C++, because this project enables no C compiler of its own.
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/opencl-stub.cpp" "${_cl_stub}")
    add_library(thermalbench_opencl_stub SHARED "${CMAKE_CURRENT_BINARY_DIR}/opencl-stub.cpp")
    set_target_properties(thermalbench_opencl_stub PROPERTIES OUTPUT_NAME OpenCL)

    # The kernels are embedded by a Python script run on the build machine. A
    # cross-build searches only the Android sysroot, so look on the host.
    # Python is asked for its own path rather than searched for: the Microsoft
    # Store install puts only zero-byte app aliases on PATH, which find_program
    # skips but which start Python when run.
    if(NOT Python3_EXECUTABLE)
        foreach(_python python3 python)
            execute_process(
                COMMAND ${_python} -c "import sys; print(sys.executable)"
                OUTPUT_VARIABLE _host_python
                OUTPUT_STRIP_TRAILING_WHITESPACE
                RESULT_VARIABLE _python_status
                ERROR_QUIET)
            if(_python_status EQUAL 0 AND NOT _host_python STREQUAL "")
                break()
            endif()
            set(_host_python "")
        endforeach()
        if(_host_python STREQUAL "")
            message(FATAL_ERROR "THERMALBENCH_OPENCL needs Python 3 on the build machine's PATH.")
        endif()
        file(TO_CMAKE_PATH "${_host_python}" _host_python)
        set(Python3_EXECUTABLE "${_host_python}" CACHE FILEPATH "" FORCE)
    endif()

    set(OpenCL_INCLUDE_DIR "${OPENCL_HEADERS_DIR}" CACHE PATH "" FORCE)
    set(OpenCL_LIBRARY thermalbench_opencl_stub CACHE STRING "" FORCE)
    set(GGML_OPENCL ON CACHE BOOL "" FORCE)
    message(STATUS "Building llama.cpp with OpenCL (headers: ${OPENCL_HEADERS_DIR})")
else()
    set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
endif()

# Every ggml backend is its own library, loaded at startup by name. A GPU
# backend that cannot load on a device (no vendor OpenCL, no Vulkan driver)
# is then skipped, instead of taking llama.cpp's CPU path down with it.
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)

add_subdirectory(${LLAMA_CPP_DIR} ${CMAKE_CURRENT_BINARY_DIR}/llama.cpp EXCLUDE_FROM_ALL)

add_library(thermalbench_llama SHARED llamacpp_jni.cpp)

# Nothing links the backend modules, so they are built because this says so.
set(_backends ggml-cpu)
if(THERMALBENCH_VULKAN)
    list(APPEND _backends ggml-vulkan)
    target_compile_definitions(thermalbench_llama PRIVATE THERMALBENCH_HAS_VULKAN)
endif()
if(THERMALBENCH_OPENCL)
    list(APPEND _backends ggml-opencl)
    target_compile_definitions(thermalbench_llama PRIVATE THERMALBENCH_HAS_OPENCL)
endif()
add_dependencies(thermalbench_llama ${_backends})

# Stamp the exact llama.cpp version into the library, so every result document
# names the runtime that produced it. Throughput differs between llama.cpp
# releases; an unrecorded version change would read as a device difference.
execute_process(
    COMMAND git -C "${LLAMA_CPP_DIR}" describe --tags --always --dirty
    OUTPUT_VARIABLE LLAMA_CPP_VERSION
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET)
if(LLAMA_CPP_VERSION STREQUAL "")
    set(LLAMA_CPP_VERSION "unknown")
endif()
if(DEFINED LLAMA_CPP_EXPECTED_TAG AND NOT LLAMA_CPP_EXPECTED_TAG STREQUAL ""
        AND NOT LLAMA_CPP_VERSION STREQUAL LLAMA_CPP_EXPECTED_TAG)
    message(WARNING
        "llama.cpp checkout is ${LLAMA_CPP_VERSION}, but the project is pinned to "
        "${LLAMA_CPP_EXPECTED_TAG}. Results will not be comparable with pinned builds.")
endif()
message(STATUS "Linking llama.cpp ${LLAMA_CPP_VERSION}")
target_compile_definitions(thermalbench_llama PRIVATE
    THERMALBENCH_LLAMA_VERSION="llama.cpp ${LLAMA_CPP_VERSION}")

target_include_directories(thermalbench_llama PRIVATE
    ${LLAMA_CPP_DIR}/include
    ${LLAMA_CPP_DIR}/ggml/include)

find_library(log-lib log)
target_link_libraries(thermalbench_llama llama ggml ${log-lib} android)
